### C1. 节点结构设计

**smolvla_node.py**：

\`\`\`python
#!/usr/bin/env python3
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image, JointState
from std_msgs.msg import String
from smolvla_msgs.msg import SmolVLAAction
from cv_bridge import CvBridge
import torch
import numpy as np
from transformers import AutoTokenizer

# 添加SmolVLA路径
import sys
sys.path.insert(0, '/path/to/lerobot/src')
from lerobot.policies.smolvla.modeling_smolvla import SmolVLAPolicy


class SmolVLANode(Node):
    def __init__(self):
        super().__init__('smolvla_node')
        
        # 参数
        self.declare_parameters(
            namespace='',
            parameters=[
                ('model_path', 'lerobot/smolvla_base'),
                ('device', 'cuda'),
                ('control_frequency', 10.0),
                ('num_cameras', 1),
                ('state_dim', 14),
                ('action_dim', 6),
            ]
        )
        
        # 获取参数
        self.model_path = self.get_parameter('model_path').value
        self.device = self.get_parameter('device').value
        self.control_freq = self.get_parameter('control_frequency').value
        self.num_cameras = self.get_parameter('num_cameras').value
        
        # 加载SmolVLA模型
        self.get_logger().info('Loading SmolVLA model...')
        self.policy = SmolVLAPolicy.from_pretrained(self.model_path)
        self.policy = self.policy.to(self.device).float().eval()
        self.get_logger().info('✅ Model loaded successfully')
        
        # 加载tokenizer
        self.tokenizer = AutoTokenizer.from_pretrained(
            "HuggingFaceTB/SmolVLM2-500M-Video-Instruct"
        )
        
        # 初始化CV Bridge
        self.bridge = CvBridge()
        
        # 数据缓存
        self.latest_image = None
        self.latest_state = None
        self.current_task = "Pick up the object"
        self.action_queue = []
        
        # 订阅器
        self.image_sub = self.create_subscription(
            Image,
            '/camera/color/image_raw',
            self.image_callback,
            10
        )
        
        self.state_sub = self.create_subscription(
            JointState,
            '/joint_states',
            self.state_callback,
            10
        )
        
        self.task_sub = self.create_subscription(
            String,
            '/task_instruction',
            self.task_callback,
            10
        )
        
        # 发布器
        self.action_pub = self.create_publisher(
            SmolVLAAction,
            '/smolvla/action',
            10
        )
        
        # 推理定时器
        self.inference_timer = self.create_timer(
            1.0 / self.control_freq,  # 10Hz
            self.inference_callback
        )
        
        self.get_logger().info('SmolVLA Node initialized')
    
    def image_callback(self, msg):
        """接收相机图像"""
        try:
            # ROS Image → OpenCV
            cv_image = self.bridge.imgmsg_to_cv2(msg, "rgb8")
            self.latest_image = cv_image
        except Exception as e:
            self.get_logger().error(f'Image conversion error: {e}')
    
    def state_callback(self, msg):
        """接收关节状态"""
        self.latest_state = np.array(msg.position)
    
    def task_callback(self, msg):
        """接收任务指令"""
        self.current_task = msg.data
        self.get_logger().info(f'Task updated: {self.current_task}')
    
    def preprocess_image(self, image):
        """图像预处理"""
        import cv2
        
        # 调整大小到256x256
        resized = cv2.resize(image, (256, 256))
        
        # 归一化到[0, 1]
        normalized = resized.astype(np.float32) / 255.0
        
        # 转为Tensor (C, H, W)
        tensor = torch.from_numpy(normalized).permute(2, 0, 1)
        
        return tensor
    
    def map_robot_state(self, joint_states):
        """Parol6状态 → SmolVLA 14维输入"""
        state = torch.zeros(14)
        state[0:6] = torch.from_numpy(joint_states)  # 关节角度
        # TODO: 添加更多状态信息
        # state[6:12] = end_effector_pose
        # state[12] = gripper_state
        # state[13] = velocity
        return state
    
    def inference_callback(self):
        """推理回调（10Hz）"""
        # 检查数据就绪
        if self.latest_image is None or self.latest_state is None:
            return
        
        try:
            # 准备输入
            batch = self.prepare_batch()
            
            # 推理
            with torch.no_grad():
                action = self.policy.select_action(batch)
            
            # 发布动作
            self.publish_action(action)
            
        except Exception as e:
            self.get_logger().error(f'Inference error: {e}')
    
    def prepare_batch(self):
        """准备模型输入"""
        # 预处理图像
        image_tensor = self.preprocess_image(self.latest_image)
        image_tensor = image_tensor.unsqueeze(0).to(self.device)  # Add batch dim
        
        # 准备状态
        state_tensor = self.map_robot_state(self.latest_state)
        state_tensor = state_tensor.unsqueeze(0).to(self.device)
        
        # 准备语言指令
        tokens = self.tokenizer(self.current_task, return_tensors="pt")
        lang_tokens = tokens['input_ids'].to(self.device)
        lang_mask = tokens['attention_mask'].to(self.device).bool()
        
        # 构建batch
        batch = {
            "observation.images.camera1": image_tensor,
            "observation.images.camera2": image_tensor.clone(),
            "observation.images.camera3": image_tensor.clone(),
            "observation.state": state_tensor,
            "observation.language.tokens": lang_tokens,
            "observation.language.attention_mask": lang_mask,
        }
        
        return batch
    
    def publish_action(self, action):
        """发布动作"""
        msg = SmolVLAAction()
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.action.data = action.squeeze(0).cpu().numpy().tolist()
        msg.confidence = 1.0  # TODO: 添加置信度计算
        
        self.action_pub.publish(msg)


def main(args=None):
    rclpy.init(args=args)
    node = SmolVLANode()
    
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()

### C2. 动作映射实现

**map_smolvla_to_parol6.py**：

\`\`\`python
import numpy as np

class ActionMapper:
    """SmolVLA动作 → Parol6指令映射"""
    
    def __init__(self):
        # Parol6关节限位
        self.joint_limits = np.array([
            [-180, 180],  # Joint 0
            [-90, 90],    # Joint 1
            [-180, 180],  # Joint 2
            [-90, 90],    # Joint 3
            [-180, 180],  # Joint 4
            [0, 1],       # Gripper
        ])
        
        # 速度限制（度/秒）
        self.max_velocity = np.array([90, 90, 90, 90, 90, 0.5])
        
        # 控制周期
        self.dt = 0.05  # 20Hz
    
    def map_action(self, smolvla_action, current_state):
        """
        映射SmolVLA输出到Parol6指令
        
        Args:
            smolvla_action: [6] 或 [7] 维数组
            current_state: [6] 当前关节状态
            
        Returns:
            parol6_command: [6] Parol6关节目标
        """
        # 选项1：直接使用（如果SmolVLA输出就是关节角度）
        target = smolvla_action[:6]
        
        # 选项2：增量控制
        # delta = smolvla_action[:6]
        # target = current_state + delta
        
        # 选项3：末端控制（需要逆运动学）
        # target = self.inverse_kinematics(smolvla_action)
        
        # 安全检查
        target = self.clip_to_limits(target)
        target = self.velocity_filter(target, current_state)
        
        return target
    
    def clip_to_limits(self, action):
        """限位裁剪"""
        clipped = np.clip(
            action,
            self.joint_limits[:, 0],
            self.joint_limits[:, 1]
        )
        return clipped
    
    def velocity_filter(self, target, current):
        """速度限制"""
        delta = target - current
        max_delta = self.max_velocity * self.dt
        
        # 限制每个关节的最大变化量
        delta_clipped = np.clip(delta, -max_delta, max_delta)
        
        filtered = current + delta_clipped
        return filtered

import numpy as np

class SafetyChecker:
    """安全检查模块"""
    
    def __init__(self):
        # 关节限位
        self.joint_limits = np.array([
            [-180, 180],  # 度
            [-90, 90],
            [-180, 180],
            [-90, 90],
            [-180, 180],
            [0, 1],
        ])
        
        # 速度限制
        self.max_velocity = np.array([90, 90, 90, 90, 90, 0.5])  # 度/秒
        
        # 工作空间限制（笛卡尔坐标）
        self.workspace_limits = {
            'x': [-0.5, 0.5],
            'y': [-0.5, 0.5],
            'z': [0.0, 0.8],
        }
    
    def check_action(self, action, current_state, dt=0.05):
        """
        验证动作安全性
        
        Returns:
            (is_safe, message)
        """
        # 1. 关节限位检查
        if not self.check_joint_limits(action):
            return False, "超出关节限位"
        
        # 2. 速度限制检查
        velocity = (action - current_state) / dt
        if not self.check_velocity_limits(velocity):
            return False, "速度过快"
        
        # 3. 工作空间检查（需要正运动学）
        # end_effector_pos = self.forward_kinematics(action)
        # if not self.check_workspace(end_effector_pos):
        #     return False, "超出工作空间"
        
        # 4. 自碰撞检查（需要碰撞检测库）
        # if self.check_self_collision(action):
        #     return False, "检测到自碰撞"
        
        return True, "安全"
    
    def check_joint_limits(self, action):
        """检查关节限位"""
        for i, (val, (min_val, max_val)) in enumerate(
            zip(action, self.joint_limits)
        ):
            if val < min_val or val > max_val:
                print(f"Joint {i}: {val} 超出范围 [{min_val}, {max_val}]")
                return False
        return True
    
    def check_velocity_limits(self, velocity):
        """检查速度限制"""
        for i, (vel, max_vel) in enumerate(
            zip(np.abs(velocity), self.max_velocity)
        ):
            if vel > max_vel:
                print(f"Joint {i}: 速度{vel:.1f}°/s 超过限制{max_vel}°/s")
                return False
        return True
    
    def check_workspace(self, position):
        """检查工作空间"""
        x, y, z = position[:3]
        
        if not (self.workspace_limits['x'][0] <= x <= self.workspace_limits['x'][1]):
            return False
        if not (self.workspace_limits['y'][0] <= y <= self.workspace_limits['y'][1]):
            return False
        if not (self.workspace_limits['z'][0] <= z <= self.workspace_limits['z'][1]):
            return False
        
        return True

#!/usr/bin/env python3
import rclpy
from rclpy.node import Node
from smolvla_msgs.msg import SmolVLAAction
import time
import numpy as np


class PerformanceMonitor(Node):
    def __init__(self):
        super().__init__('performance_monitor')
        
        self.latencies = []
        self.last_time = None
        
        self.sub = self.create_subscription(
            SmolVLAAction,
            '/smolvla/action',
            self.callback,
            10
        )
        
        self.timer = self.create_timer(10.0, self.print_stats)
    
    def callback(self, msg):
        now = time.time()
        
        if self.last_time is not None:
            latency = (now - self.last_time) * 1000  # ms
            self.latencies.append(latency)
        
        self.last_time = now
    
    def print_stats(self):
        if len(self.latencies) == 0:
            return
        
        self.get_logger().info(
            f"性能统计（最近{len(self.latencies)}次）:\n"
            f"  平均延迟: {np.mean(self.latencies):.1f} ms\n"
            f"  最小延迟: {np.min(self.latencies):.1f} ms\n"
            f"  最大延迟: {np.max(self.latencies):.1f} ms\n"
            f"  频率: {1000/np.mean(self.latencies):.1f} Hz"
        )
        
        self.latencies = []


def main():
    rclpy.init()
    node = PerformanceMonitor()
    rclpy.spin(node)


if __name__ == '__main__':
    main()

class ActionFilter:
    def __init__(self, alpha=0.3):
        self.alpha = alpha
        self.last_action = None
    
    def filter(self, action):
        if self.last_action is None:
            self.last_action = action
            return action
        
        # 指数移动平均
        filtered = self.alpha * action + (1 - self.alpha) * self.last_action
        self.last_action = filtered
        return filtered
